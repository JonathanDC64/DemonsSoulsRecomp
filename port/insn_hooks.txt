# des_cut_cam (port/des_options.cpp): the cutscene camera's keys (sibcam).
# 0x005F71D0 loads the nearest of two keys 1/30 s apart (s < 0.5 ? earlier :
# later); with DES_CUTFPS=1 the hooks blend the two by s. The first decides
# once a call; the other three replace the loaded vector.
0x005F7254=des_cut_cam   # lwz r9, -0x7F6C(r30): s (f10), the gap (f11) and the key indices are set; decides once a call
0x005F72EC=des_cut_cam   # lvx v4: position (earlier r1+0xA0, later r1+0xB0)
0x005F7394=des_cut_cam   # lvx v12: rotation in radians, before the wrap (earlier r1+0xD0, later r1+0xA0)
0x005F743C=des_cut_cam   # lvx v2: scale (earlier r1+0x70, later r1+0xA0)

# des_mouse_look (port/des_options.cpp): mouse look (docs/41) in the ChrExFollowCam
# update. The update works yaw and pitch out again from the eye and the target
# every frame, so the mouse's turn goes in where the stick's does, on a frame
# that counts as one where the stick moved.
0x00119670=des_mouse_look   # li r0, 4080: the entry (r3 = the camera); takes the counts once a game frame
0x0011C910=des_mouse_look   # lfs f0, 0xF0(r1): the stick's pitch input, tested against 0 next; nonzero with a turn to make
0x0011C758=des_mouse_look   # lwz r9, -0x7F84(r30): the angles rebuilt, the stick's turn next; adds the mouse's turn
# The bow's aim view (ChrAimCam, the camera manager's modes 1 and 2) keeps its
# angles from frame to frame, so the mouse's turn goes onto them at its entry.
0x00110608=des_mouse_look   # li r0, 1776: the aim camera's entry (r3 = the camera, r5 = the character); takes the counts and turns its kept angles
