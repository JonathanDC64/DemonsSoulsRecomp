# Instruction hooks for the PPU lift: ppu_lifter.py --insn-hooks @port/insn_hooks.txt
# Each line: guest address = host function. The lifted code calls the function
# with (ctx, address) right after that instruction. See docs/28, "Camera
# catch-up at any frame rate".
#
# des_motion_blur (port/des_options.cpp): DES_MOTION_BLUR=0 inverts the blur
# stage's test of its enable byte, like RPCS3's "Disable Motion Blur" patch
# (bne -> beq at 0x00217E7C).
0x00217E70=des_motion_blur   # cmpwi cr7, r0 (= byte +0x1D4), 0

# des_cut_cam (port/des_options.cpp): the cutscene camera's keys (sibcam).
# 0x005F71D0 loads the nearest of two keys 1/30 s apart (s < 0.5 ? earlier :
# later); with DES_CUTFPS=1 the hooks blend the two by s. The first decides
# once a call; the other three replace the loaded vector.
0x005F7254=des_cut_cam   # lwz r9, -0x7F6C(r30): s (f10), the gap (f11) and the key indices are set; decides once a call
0x005F72EC=des_cut_cam   # lvx v4: position (earlier r1+0xA0, later r1+0xB0)
0x005F7394=des_cut_cam   # lvx v12: rotation in radians, before the wrap (earlier r1+0xD0, later r1+0xA0)
0x005F743C=des_cut_cam   # lvx v2: scale (earlier r1+0x70, later r1+0xA0)

# des_mouse_look (port/des_options.cpp): mouse look (docs/41). The entry of the
# ChrExFollowCam update, after its first instruction: r3 = the camera, f1 = dt.
# The hook adds the mouse's turn to the camera's yaw and pitch before the update
# turns and places the camera, so they agree in the same frame.
0x00119670=des_mouse_look   # li r0, 4080

# des_cam_rate (port/des_options.cpp): loads of a per-frame ease factor k in the
# camera code (cur += (target - cur) * k, tuned for 30 fps). Each is an lfs; the
# hook rescales k to the frame time.

# ChrExFollowCam update (0x00119670), the usual third-person camera.
# All of these run every frame while walking.
0x00119818=des_cam_rate   # k = +0x294 (0.1)
0x001198D8=des_cam_rate   # k = 0.1 (constant), +0x174 towards +0x178
0x00119FAC=des_cam_rate   # k = +0x144
0x0011A018=des_cam_rate   # k = +0x144
0x0011B830=des_cam_rate   # k = +0x1A8 (0.4)
0x0011C0E8=des_cam_rate   # k = 1/30 (constant), +0x17C back towards +0x180
0x0011C2BC=des_cam_rate   # k = +0x17C (0.1): the follow point's chase (vector lerp)
0x0011D6E4=des_cam_rate   # k = +0x220 (0.06), +0x130
0x0011D820=des_cam_rate   # k = +0x220 (0.06), +0x130

# The manager's second camera (0x00110608).
0x00110808=des_cam_rate   # k = +0x138, +0x130 towards +0x134

# des_dof (port/des_options.cpp): DES_DOF=0 zeroes the two depth-of-field
# strengths that the image-process update blends from the area's DOF_BANK rows.
0x002184DC=des_dof   # lbz r0, 0x1C4(r31), the first instruction after the blend (bl 0x005EAF18)

# ChrFollowCam update (0x00122C40).
0x00123100=des_cam_rate   # k = +0x114
0x00123164=des_cam_rate   # k = +0x114
0x00124188=des_cam_rate   # k = +0x158
0x0012554C=des_cam_rate   # k = +0x19C
0x001256D4=des_cam_rate   # k = +0x19C

# des_slide_rate (port/des_options.cpp): a character proxy's slide on slopes
# too steep to stand on, in its velocity update 0x001FE778. The gain is added
# and the decay applied once a frame, tuned for 30 fps.
0x001FEAF4=des_slide_rate   # decay 0.65 (0x019D9C40), slide *= decay
0x001FED50=des_slide_rate   # gain 1.0 (0x019D9C3C), slide += gravity along the slope * gain

# des_viewer_rate (port/des_options.cpp): the character creation previews'
# stick steps add the right stick to the yaw and pitch once a frame, tuned
# for 30 fps. The hook scales the stick (f1, f2) by the frame's 30 Hz steps.
0x008DA0E0=des_viewer_rate   # stdu r1, -0x290(r1), the first instruction: the face (Edit Appearance)
0x008DB170=des_viewer_rate   # stdu r1, -0x290(r1), the first instruction: the body
